// ===================================================================
// PROGRAMA PRINCIPAL - Configuración de la API
// ===================================================================
// Este archivo configura toda la aplicación ASP.NET Core Web API
// Es como un "director de orquesta" que organiza cómo funciona todo
using AlumnosApi.Data;  // Importa el contexto de la base de datos
using Microsoft.EntityFrameworkCore;  // Importa Entity Framework para trabajar con BD
// ===================================================================
// PASO 1: Crear el "constructor" de la aplicación
// ===================================================================
// CreateBuilder(args) prepara los servicios que necesitará nuestra API
// Como si estuviéramos preparando los instrumentos de la orquesta
var builder = WebApplication.CreateBuilder(args);
// ===================================================================
// PASO 2: Registrar los servicios en el contenedor de inyección 
// ===================================================================
// Aquí le decimos a la aplicación QUÉ debe usar
// 📊 Conectar con la base de datos SQL Server
// - AddDbContext: Registra nuestro "AlumnosContext" (el intermediario con la BD)
// - UseSqlServer: Especifica que usamos SQL Server (no otra BD)
// - GetConnectionString("Alumnos"): Obtiene la conexión del archivo appsettings.json
builder.Services.AddDbContext<AlumnosContext>(opt =>
    opt.UseSqlServer(builder.Configuration.GetConnectionString("Alumnos")));
// 🎮 Registrar los controladores (las "rutas" del API)
// Los controladores reciben las peticiones HTTP (GET, POST, PUT, DELETE)
builder.Services.AddControllers();
// 📖 Explorer API (Swagger)
// - AddEndpointsApiExplorer: Detecta automáticamente nuestros endpoints
// - AddSwaggerGen: Genera la documentación interactiva
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();
// ===================================================================
// PASO 3: Crear la aplicación con las configuraciones anteriores
// ===================================================================
// Aquí se construye la aplicación con TODOS los servicios registrados
// Es como ensamblar la orquesta lista para tocar
var app = builder.Build();
// ===================================================================
// PASO 4: Configurar el "middleware" (procesadores de peticiones)
// ===================================================================
// El middleware procesa cada petición HTTP en este orden:
// 🔒 Redirección HTTPS
// Si alguien intenta conectar por HTTP, lo redirige automáticamente a HTTPS
// Esto mejora la seguridad de la comunicación
// app.UseHttpsRedirection(); //y bloquea el celular conectado por usb
// 📖 Activar Swagger
// - UseSwagger: Genera el archivo JSON con la documentación
// - UseSwaggerUI: Muestra la interfaz web en /swagger
app.UseSwagger();
app.UseSwaggerUI();
// ===================================================================
// PASO 5: Configurar las rutas del API
// ===================================================================
// 🏠 Ruta raíz (cuando accedes a https://localhost:50554/)
// Redirige automáticamente a /swagger para ver la documentación
app.MapGet("/", () => Results.Redirect("/swagger"));
// 🎮 Mapear todos los controladores
// Aquí se registran todas las rutas definidas en los Controllers
// (AlumnosController, AsignaturasController, NotasController)
app.MapControllers();
// ===================================================================
// PASO 6: Iniciar la aplicación
// ===================================================================
// Run() inicia el servidor y espera peticiones HTTP
// La aplicación queda corriendo hasta que la cierres
app.Run();

using AlumnosApi.Models;
using Microsoft.EntityFrameworkCore;
namespace AlumnosApi.Data;
// ===================================================================
// CONTEXTO DE DATOS: AlumnosContext
// ===================================================================
// Este es el "intermediario" entre nuestra aplicación y la base de datos
// Entity Framework usa esta clase para leer y escribir en SQL Server
// Es como un "intérprete" que traduce nuestro código C# a comandos SQL
public class AlumnosContext : DbContext
{
    // Constructor
    // DbContextOptions contiene la configuración de conexión
    // Se pasa automáticamente desde Program.cs
    public AlumnosContext(DbContextOptions<AlumnosContext> options) : base(options) { }
    // DbSet: Tablas de la base de datos
    // Cada DbSet representa una tabla en la BD
    // Set<Alumno>() obtiene la colección de alumnos de la BD
    /// Tabla ALUMNOS - Contiene todos los alumnos registrados
    /// Accedemos así: _db.Alumnos.ToListAsync() para listar, _db.Alumnos.Add() para agregar
    public DbSet<Alumno> Alumnos => Set<Alumno>();
    /// Tabla ASIGNATURAS - Contiene todas las asignaturas disponibles
    /// Ejemplo: Matemática, Lenguaje, Historia, Inglés, etc.
    public DbSet<Asignatura> Asignaturas => Set<Asignatura>();
    /// Tabla NOTAS - Contiene las calificaciones de cada alumno por asignatura
    /// Relaciona: Qué alumno (CodAlumno) tuvo qué nota (Valor) en qué asignatura (IdAsignatura)
    public DbSet<Nota> Notas => Set<Nota>();
    // OnModelCreating: Configuración del mapeo ORM
    // Aquí definimos la relación entre nuestros objetos C# y las tablas SQL
    // Es como decir: "Este campo C# se llama NOMBRES en la tabla SQL"
    // IMPORTANTE: Los nombres de tabla DEBEN coincidir exactamente con los de la BD
    protected override void OnModelCreating(ModelBuilder mb)
    {
        // Configuración de tabla ALUMNOS
        mb.Entity<Alumno>(e =>
        {
            // Especificar el nombre exacto de la tabla en SQL Server
            e.ToTable("ALUMNOS");
            // Definir la clave primaria (no puede repetirse)
            e.HasKey(x => x.CodAlumno);
            // Mapear cada propiedad de C# a su columna en SQL
            // HasColumnName("COD_ALUMNO") = la columna en BD se llama "COD_ALUMNO"
            // ValueGeneratedNever() = no se genera automáticamente (la asignamos nosotros)
            e.Property(x => x.CodAlumno)
                .HasColumnName("COD_ALUMNO")
                .ValueGeneratedNever();
            // Mapear el campo Nombres a su columna NOMBRES en la BD
            e.Property(x => x.Nombres)
                .HasColumnName("NOMBRES");
            // Mapear el campo Apellidos a su columna APELLIDOS en la BD
            e.Property(x => x.Apellidos)
                .HasColumnName("APELLIDOS");
        });
        // Configuración de tabla ASIGNATURAS
        mb.Entity<Asignatura>(e =>
        {
            e.ToTable("ASIGNATURAS");
            // IdAsignatura es la clave primaria
            // Sin ValueGeneratedNever() porque SQL Server la genera automáticamente
            e.HasKey(x => x.IdAsignatura);
            e.Property(x => x.IdAsignatura)
                .HasColumnName("ID_ASIGNATURA");
            e.Property(x => x.NomAsignatura)
                .HasColumnName("NOM_ASIGNATURA");
        });
        // Configuración de tabla NOTAS
        mb.Entity<Nota>(e =>
        {
            e.ToTable("NOTAS");
            // IdNota es la clave primaria de la tabla de notas
            e.HasKey(x => x.IdNota);
            e.Property(x => x.IdNota)
                .HasColumnName("ID_NOTA");
            // Valor: la calificación (0 a 10, con un decimal)
            // HasColumnType("decimal(2,1)") = máximo 2 dígitos, 1 decimal (Ej: 9.5)
            e.Property(x => x.Valor)
                .HasColumnName("NOTA")
                .HasColumnType("decimal(2,1)");
            // Número de nota para el alumno en esa asignatura (1era, 2da, 3era nota)
            e.Property(x => x.NNota)
                .HasColumnName("N_NOTA");
            // Relación con Alumno: qué alumno tiene esta nota
            e.Property(x => x.CodAlumno)
                .HasColumnName("COD_ALUMNO");
            // Relación con Asignatura: en qué asignatura es esta nota
            e.Property(x => x.IdAsignatura)
                .HasColumnName("ID_ASIGNATURA");
        });
    }
}

using AlumnosApi.Data;  // Importa el contexto de BD
using AlumnosApi.Models;  // Importa los modelos (Alumno, etc.)
using Microsoft.AspNetCore.Mvc;  // Importa controlador base y atributos
using Microsoft.EntityFrameworkCore;  // Importa funciones async de EF Core
namespace AlumnosApi.Controllers;
// ===================================================================
// CONTROLADOR: AlumnosController
// ===================================================================
// Este controlador gestiona todas las operaciones relacionadas con Alumnos
// Recibe peticiones HTTP y responde con datos JSON
// Las rutas serán: /api/alumnos (por la ruta abajo)
// 🎖️ [ApiController] - Indica que esta es un controlador de API
// Activa comportamientos automáticos como validación de modelos
[ApiController]
// 🛣️ [Route("api/[controller]")] - Define la ruta base
// [controller] se reemplaza automáticamente por "Alumnos"
// Resultado: las peticiones van a /api/Alumnos
[Route("api/[controller]")]
// Hereda de ControllerBase (no ControllerBase que es para vistas HTML)
public class AlumnosController : ControllerBase
{
    // ===================================================================
    // Inyección de dependencias
    // ===================================================================
    // _db es la conexión con la base de datos
    // "readonly" significa que no puede cambiar después de asignarse
    // (es seguro y buena práctica)
    private readonly AlumnosContext _db;
    // ===================================================================
    // Constructor
    // ===================================================================
    // Se ejecuta cuando se crea una instancia del controlador
    // AlumnosContext se inyecta automáticamente desde Program.cs
    // La flecha "=>" es expresión lambda (forma corta de asignar)
    public AlumnosController(AlumnosContext db) => _db = db;
    // ===================================================================
    // MÉTODO 1: Listar todos los alumnos
    // ===================================================================
    // GET /api/alumnos
    // Obtiene TODOS los alumnos de la base de datos
    // 🔍 [HttpGet] - Ve peticiones HTTP GET
    // GET es para LEER datos (no modifica nada)
    [HttpGet]
    // async Task<IActionResult> - Método asincrónico (no bloquea el servidor)
    // IActionResult - Puede devolver distintos tipos de respuesta (Ok, BadRequest, etc.)
    public async Task<IActionResult> listarAlumnos()
    {
        // 📋 OrderBy: Ordena los alumnos por código (menor a mayor)
        // ToListAsync: Ejecuta la consulta de forma asincrónica (eficiente)
        var lista = await _db.Alumnos.OrderBy(a => a.CodAlumno).ToListAsync();
        // 200 OK - Devuelve la lista en formato JSON
        // Si está vacía, devuelve una lista vacía []
        return Ok(lista);
    }
    // ===================================================================
    // MÉTODO 2: Agregar un nuevo alumno
    // ===================================================================
    // POST /api/alumnos
    // Recibe datos JSON de un alumno nuevo y lo guarda en la BD
    // 📝 [HttpPost] - Recibe peticiones HTTP POST
    // POST es para CREAR nuevos datos
    [HttpPost]
    // El parámetro "alumno" recibe el JSON automáticamente
    // ASP.NET Core deserializa el JSON a un objeto Alumno
    public async Task<IActionResult> agregarAlumno(Alumno alumno)
    {
        // ===================================================================
        // VALIDACIONES - Verificar que los datos sean correctos
        // ===================================================================
        // No queremos guarbar datos "basura" en la BD
        // ❌ Validación 1: El código debe ser mayor a 0
        // Un alumno sin código no tiene sentido
        if (alumno.CodAlumno <= 0)
            // 400 Bad Request - El cliente envió datos inválidos
            return BadRequest("El código de alumno debe ser mayor a 0.");
        // ❌ Validación 2: Nombres y Apellidos son obligatorios
        // string.IsNullOrWhiteSpace: verifica si está vacío o solo tiene espacios
        if (string.IsNullOrWhiteSpace(alumno.Nombres) || string.IsNullOrWhiteSpace(alumno.Apellidos))
            return BadRequest("Nombres y apellidos son obligatorios.");
        // ❌ Validación 3: No puede haber dos alumnos con el mismo código
        // AnyAsync: devuelve true si existe al menos uno que cumple la condición
        if (await _db.Alumnos.AnyAsync(a => a.CodAlumno == alumno.CodAlumno))
            // 409 Conflict - Ya existe un recurso con esa identificación
            return Conflict($"Ya existe un alumno con código {alumno.CodAlumno}.");
        // ===================================================================
        // LIMPIAR DATOS - Preparar antes de guardar
        // ===================================================================
        // Trim(): elimina espacios al inicio y final
        // Ejemplo: "  Juan  " se convierte en "Juan"
        alumno.Nombres = alumno.Nombres.Trim();
        alumno.Apellidos = alumno.Apellidos.Trim();
        // ===================================================================
        // GUARDAR EN LA BASE DE DATOS
        // ===================================================================
        // Add(): añade el alumno a la lista de cambios pendientes
        // (aún NO se guarda en la BD)
        _db.Alumnos.Add(alumno);
        // SaveChangesAsync(): ejecuta todos los cambios en la BD de forma asincrónica
        // Aquí es cuando se ejecuta el INSERT en SQL Server
        await _db.SaveChangesAsync();
        // 200 OK - El alumno se guardó exitosamente
        // Devuelve el objeto alumno que se acaba de crear (con sus datos)
        return Ok(alumno);
    }
    [HttpPut]
    public async Task<IActionResult> editarAlumno(Alumno alumno)
    {
        if (string.IsNullOrWhiteSpace(alumno.Nombres) || string.IsNullOrWhiteSpace(alumno.Apellidos))
            return BadRequest("Nombres y apellidos son obligatorios.");
        var existente = await _db.Alumnos.FirstOrDefaultAsync(a => a.CodAlumno == alumno.CodAlumno);
        if (existente == null)
            return NotFound($"No existe un alumno con código {alumno.CodAlumno}.");
        existente.Nombres = alumno.Nombres.Trim();
        existente.Apellidos = alumno.Apellidos.Trim();
        await _db.SaveChangesAsync();
        return Ok(existente);
    }
}

using AlumnosApi.Data;
using AlumnosApi.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
namespace AlumnosApi.Controllers;
[ApiController]
[Route("api/[controller]")]
public class AsignaturasController : ControllerBase
{
    private readonly AlumnosContext _db;
    public AsignaturasController(AlumnosContext db) => _db = db;
    // GET /api/asignaturas - Lista todas las asignaturas
    [HttpGet]
    public async Task<IActionResult> Listar()
    {
        var lista = await _db.Asignaturas.OrderBy(a => a.IdAsignatura).ToListAsync();
        return Ok(lista);
    }
    // POST /api/asignaturas - Crea una asignatura (ID autogenerado)
    [HttpPost]
    public async Task<IActionResult> Crear(Asignatura asignatura)
    {
        if (string.IsNullOrWhiteSpace(asignatura.NomAsignatura))
            return BadRequest("El nombre de la asignatura es obligatorio.");
        var nombre = asignatura.NomAsignatura.Trim();
        if (await _db.Asignaturas.AnyAsync(a => a.NomAsignatura == nombre))
            return Conflict("Ya existe una asignatura con ese nombre.");
        asignatura.IdAsignatura = 0;
        asignatura.NomAsignatura = nombre;
        _db.Asignaturas.Add(asignatura);
        await _db.SaveChangesAsync();
        return Ok(asignatura);
    }
    [HttpPut]
    public async Task<IActionResult> Editar(Asignatura asignatura)
    {
        if (string.IsNullOrWhiteSpace(asignatura.NomAsignatura))
            return BadRequest("El nombre de la asignatura es obligatorio.");
        var nombre = asignatura.NomAsignatura.Trim();
        var existente = await _db.Asignaturas.FirstOrDefaultAsync(a => a.IdAsignatura == asignatura.IdAsignatura);
        if (existente == null)
            return NotFound("La asignatura no existe.");
        if (await _db.Asignaturas.AnyAsync(a => a.NomAsignatura == nombre && a.IdAsignatura != asignatura.IdAsignatura))
            return Conflict("Ya existe una asignatura con ese nombre.");
        existente.NomAsignatura = nombre;
        await _db.SaveChangesAsync();
        return Ok(existente);
    }
}

using AlumnosApi.Data;
using AlumnosApi.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
namespace AlumnosApi.Controllers;
[ApiController]
[Route("api/[controller]")]
public class NotasController : ControllerBase
{
    private readonly AlumnosContext _db;
    public NotasController(AlumnosContext db) => _db = db;
    // Validaciones comunes de una nota
    private static string? Validar(Nota nota)
    {
        if (nota.NNota == null || nota.NNota < 1 || nota.NNota > 3)
            return "N_NOTA debe ser 1, 2 o 3.";
        if (nota.Valor == null || nota.Valor < 1.0m || nota.Valor > 7.0m)
            return "La nota debe estar entre 1.0 y 7.0.";
        if (decimal.Round(nota.Valor.Value, 1) != nota.Valor.Value)
            return "La nota admite solo un decimal.";
        return null;
    }
    // POST /api/notas - Ingresar una nota nueva
    [HttpPost]
    public async Task<IActionResult> Ingresar(Nota nota)
    {
        var error = Validar(nota);
        if (error != null) return BadRequest(error);
        if (!await _db.Alumnos.AnyAsync(a => a.CodAlumno == nota.CodAlumno))
            return NotFound("El alumno no existe.");
        if (!await _db.Asignaturas.AnyAsync(a => a.IdAsignatura == nota.IdAsignatura))
            return NotFound("La asignatura no existe.");
        var existe = await _db.Notas.AnyAsync(n =>
            n.CodAlumno == nota.CodAlumno &&
            n.IdAsignatura == nota.IdAsignatura &&
            n.NNota == nota.NNota);
        if (existe)
            return Conflict("Esa nota ya fue ingresada. Use la edición para modificarla.");
        nota.IdNota = 0;
        _db.Notas.Add(nota);
        await _db.SaveChangesAsync();
        return Ok(nota);
    }
    // PUT /api/notas - Editar el valor de una nota existente
    [HttpPut]
    public async Task<IActionResult> Editar(Nota nota)
    {
        var error = Validar(nota);
        if (error != null) return BadRequest(error);
        var existente = await _db.Notas.FirstOrDefaultAsync(n =>
            n.CodAlumno == nota.CodAlumno &&
            n.IdAsignatura == nota.IdAsignatura &&
            n.NNota == nota.NNota);
        if (existente == null)
            return NotFound("No existe esa nota para el alumno y la asignatura indicados.");
        existente.Valor = nota.Valor;
        await _db.SaveChangesAsync();
        return Ok(existente);
    }
    // GET /api/notas/listado - Alumnos con sus notas y promedio
    [HttpGet("listado")]
    public async Task<IActionResult> Listado()
    {
        var filas = await (
            from n in _db.Notas
            join a in _db.Alumnos on n.CodAlumno equals a.CodAlumno
            join s in _db.Asignaturas on n.IdAsignatura equals s.IdAsignatura
            select new
            {
                n.CodAlumno,
                a.Nombres,
                a.Apellidos,
                n.IdAsignatura,
                s.NomAsignatura,
                n.NNota,
                n.Valor
            }).ToListAsync();
        var resultado = new List<Dictionary<string, object?>>();
        var grupos = filas
            .GroupBy(f => new { f.CodAlumno, f.IdAsignatura })
            .OrderBy(g => g.Key.CodAlumno)
            .ThenBy(g => g.First().NomAsignatura);
        foreach (var g in grupos)
        {
            var primera = g.First();
            var n1 = g.FirstOrDefault(x => x.NNota == 1);
            var n2 = g.FirstOrDefault(x => x.NNota == 2);
            var n3 = g.FirstOrDefault(x => x.NNota == 3);
            var fila = new Dictionary<string, object?>();
            fila["COD_ALUMNO"] = primera.CodAlumno;
            fila["NOMBRE_COMPLETO"] = (primera.Nombres + " " + primera.Apellidos).Trim();
            fila["NOM_ASIGNATURA"] = primera.NomAsignatura;
            // Solo se agregan los campos de las notas que existen
            if (n1 != null) { fila["N_NOTA_1"] = n1.NNota; fila["NOTA_1"] = n1.Valor; }
            if (n2 != null) { fila["N_NOTA_2"] = n2.NNota; fila["NOTA_2"] = n2.Valor; }
            if (n3 != null) { fila["N_NOTA_3"] = n3.NNota; fila["NOTA_3"] = n3.Valor; }
            // El promedio solo si están las 3 notas
            if (n1?.Valor != null && n2?.Valor != null && n3?.Valor != null)
                fila["PROMEDIO"] = Math.Round((n1.Valor.Value + n2.Valor.Value + n3.Valor.Value) / 3m, 1);
            resultado.Add(fila);
        }
        return Ok(resultado);
    }

-- baseDatosAlumnos_creacion.sql
-- Crea la base de datos baseDatosAlumnos y las tablas ALUMNOS, ASIGNATURAS y NOTAS.
-- Los nombres de tablas y columnas coinciden con AlumnosContext.cs (tablas en plural).
-- Es re-ejecutable: no borra nada y solo crea lo que no existe.
-- No usa rutas de archivo fijas, por lo que funciona en cualquier equipo con SQL Server, Express o LocalDB.
USE [master]
GO
-- Crea la base de datos solo si no existe. SQL Server elige la carpeta de datos por defecto.
IF DB_ID(N'baseDatosAlumnos') IS NULL
    CREATE DATABASE [baseDatosAlumnos]
GO
USE [baseDatosAlumnos]
GO
-- Tabla ALUMNOS: COD_ALUMNO lo ingresa el usuario (no es autonumerico, igual que ValueGeneratedNever en el contexto).
IF OBJECT_ID(N'dbo.ALUMNOS', N'U') IS NULL
CREATE TABLE [dbo].[ALUMNOS](
    [COD_ALUMNO] [int] NOT NULL,
    [NOMBRES]    [varchar](50) NOT NULL,
    [APELLIDOS]  [varchar](50) NOT NULL,
    CONSTRAINT [PK_ALUMNOS] PRIMARY KEY CLUSTERED ([COD_ALUMNO] ASC)
)
GO
-- Tabla ASIGNATURAS: ID_ASIGNATURA es autonumerico y el nombre no se puede repetir (la API tambien lo valida).
IF OBJECT_ID(N'dbo.ASIGNATURAS', N'U') IS NULL
CREATE TABLE [dbo].[ASIGNATURAS](
    [ID_ASIGNATURA]  [int] IDENTITY(1,1) NOT NULL,
    [NOM_ASIGNATURA] [varchar](50) NOT NULL,
    CONSTRAINT [PK_ASIGNATURAS] PRIMARY KEY CLUSTERED ([ID_ASIGNATURA] ASC),
    CONSTRAINT [UQ_ASIGNATURAS_NOMBRE] UNIQUE ([NOM_ASIGNATURA])
)
GO
-- Tabla NOTAS: una fila por cada nota (hasta 3 por alumno y asignatura).
-- N_NOTA indica cual de las 3 notas es (1, 2 o 3). NOTA es decimal(2,1): un solo decimal.
IF OBJECT_ID(N'dbo.NOTAS', N'U') IS NULL
CREATE TABLE [dbo].[NOTAS](
    [ID_NOTA]        [int] IDENTITY(1,1) NOT NULL,
    [NOTA]           [decimal](2,1) NOT NULL,
    [N_NOTA]         [int] NOT NULL,
    [COD_ALUMNO]     [int] NOT NULL,
    [ID_ASIGNATURA]  [int] NOT NULL,
    CONSTRAINT [PK_NOTAS] PRIMARY KEY CLUSTERED ([ID_NOTA] ASC),
    -- Relaciones: toda nota pertenece a un alumno y a una asignatura existentes.
    CONSTRAINT [FK_NOTAS_ALUMNOS] FOREIGN KEY ([COD_ALUMNO]) REFERENCES [dbo].[ALUMNOS]([COD_ALUMNO]),
    CONSTRAINT [FK_NOTAS_ASIGNATURAS] FOREIGN KEY ([ID_ASIGNATURA]) REFERENCES [dbo].[ASIGNATURAS]([ID_ASIGNATURA]),
    -- Reglas del enunciado: N_NOTA entre 1 y 3, y nota entre 1.0 y 7.0.
    CONSTRAINT [CK_NOTAS_N_NOTA] CHECK ([N_NOTA] BETWEEN 1 AND 3),
    CONSTRAINT [CK_NOTAS_NOTA] CHECK ([NOTA] BETWEEN 1.0 AND 7.0),
    -- Impide repetir la misma nota (N_NOTA) para un mismo alumno y asignatura, incluso con peticiones simultaneas.
    CONSTRAINT [UQ_NOTAS_ALUMNO_ASIGNATURA_NNOTA] UNIQUE ([COD_ALUMNO], [ID_ASIGNATURA], [N_NOTA])
)
GO
