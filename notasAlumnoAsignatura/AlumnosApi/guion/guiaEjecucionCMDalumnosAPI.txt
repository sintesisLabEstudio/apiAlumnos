# GUÍA Y COMANDOS COMPLETOS PARA EJECUTAR LA API SIN VISUAL STUDIO
# Se usa UNA SOLA instancia de SQL Server: LocalDB, la misma de appsettings.json.
# No adjuntar el .mdf a SQLEXPRESS: dos instancias abriendo el mismo archivo se bloquean entre sí.

# 1. Verificar los archivos físicos de la base de datos en la carpeta BD
dir "C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD"

# 2. Crear la base de datos ALUMNOS con el script original del profesor (NO modificar el script)
sqlcmd -S "(localdb)\MSSQLLocalDB" -E -i "C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\Data\ALUMNOS.sql"

# 3. Comprobar la existencia de las tablas (ALUMNO, ASIGNATURA, NOTAS) en la base de datos ALUMNOS
sqlcmd -S "(localdb)\MSSQLLocalDB" -E -d ALUMNOS -Q "SELECT name FROM sys.tables"

# 4. Comprobar el contenido de registros en la tabla ALUMNO
sqlcmd -S "(localdb)\MSSQLLocalDB" -E -d ALUMNOS -Q "SELECT COUNT(*) FROM ALUMNO"

# 5. (Opcional) Crear un respaldo de seguridad (.bak)
sqlcmd -S "(localdb)\MSSQLLocalDB" -E -Q "BACKUP DATABASE ALUMNOS TO DISK='C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD\respaldo_ALUMNOS.bak'"

# 6. Moverse al directorio de la API y ejecutarla en HTTP por el puerto 50555
# El puerto 50555 es el que consume la app Android (RetrofitClient.BASE_URL).
# No se necesita certificado HTTPS porque UseHttpsRedirection está desactivado en Program.cs.
cd C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi

dotnet run --urls http://localhost:50555

Abre 
http://localhost:50555/swagger
