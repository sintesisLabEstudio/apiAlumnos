# GUÍA Y COMANDOS COMPLETOS PARA EJECUTAR LA API SIN VISUAL STUDIO

# 1. Verificar los archivos físicos de la base de datos en la carpeta BD
dir "C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD"

# 2. Otorgar permisos de control total a SQL Server Express sobre la carpeta BD (PowerShell)

icacls "C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD" /grant 'NT SERVICE\MSSQL$SQLEXPRESS:(OI)(CI)F'

# 3. Adjuntar la base de datos existente (.mdf y _log.ldf) en SQL Server Express

sqlcmd -S localhost\SQLEXPRESS -E -Q "CREATE DATABASE baseDatosAlumnos ON (FILENAME='C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD\baseDatosAlumnos.mdf'), (FILENAME='C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD\baseDatosAlumnos_log.ldf') FOR ATTACH"

# 4. Crear un respaldo de seguridad (.bak) preventivo de la base adjuntada

sqlcmd -S localhost\SQLEXPRESS -E -Q "BACKUP DATABASE baseDatosAlumnos TO DISK='C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD\respaldo_baseDatosAlumnos1.bak'"

# 5. Verificar que el archivo de respaldo quedó íntegro y válido

sqlcmd -S localhost\SQLEXPRESS -E -Q "RESTORE VERIFYONLY FROM DISK='C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi\BD\respaldo_baseDatosAlumnos1.bak'"

# 6. Comprobar la existencia de las tablas en la base de datos

sqlcmd -S localhost\SQLEXPRESS -E -d baseDatosAlumnos -Q "SELECT name FROM sys.tables"

# 7. Comprobar el contenido de registros en la tabla ALUMNOS

sqlcmd -S localhost\SQLEXPRESS -E -d baseDatosAlumnos -Q "SELECT COUNT(*) FROM ALUMNOS"

# 8. Moverse al directorio de la API y ejecutarla escuchando en todas las interfaces de red

cd C:\desarrollo\tnt\notasAlumnoAsignatura\AlumnosApi
